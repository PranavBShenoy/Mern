const http = require('http')
const hostname = '127.0.0.1'
const port = 3000

//Create server
const server = http.createServer((req, res) => {
res.setHeader('Content-Type', 'text/html') // Indicates the type of data sent to Client
if (req.url === '/') {
res.statusCode = 200 // Status Code OK
res.end(`<h1> Hello World </h1>
       <p> Welcome to Home page </p>`) // End the Response with Welcome to the
homepage!
} else if (req.url === '/about') {
res.statusCode = 200
res.end(` <!DOCTYPE html>
        <html>
        <head>
        <title> About page </title>
        </head>
        <body>
       <h1> About Page </h1>
       <p> This is About Page </p>
       </body>
       </html> `)
} else {
res.statusCode = 404; // Status Code Not Found!
res.end(`<h1> Page not found </h1>`)
}
})

// START the server.
server.listen(port, hostname, () => {
console.log(`Server running at http://ℎ𝑜𝑠𝑡𝑛𝑎𝑚𝑒:{port}/`)
})
